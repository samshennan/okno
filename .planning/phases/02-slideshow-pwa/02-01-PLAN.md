  <action>
Create `public/sw.js` implementing network-first strategy for photos, cache-first for static assets:

```javascript
const CACHE_VERSION = 'okno-v1';
const PHOTO_CACHE = 'okno-photos-v1';

const STATIC_ASSETS = [
  '/slideshow.html',
  '/manifest.json',
  '/icons/icon-192.png',
  '/icons/icon-512.png',
  '/icons/apple-touch-icon.png'
];

// Install: precache static assets
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_VERSION).then((cache) => {
      return cache.addAll(STATIC_ASSETS);
    })
  );
  self.skipWaiting(); // Activate immediately
});

// Activate: clean up old caches
self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((cacheNames) => {
      return Promise.all(
        cacheNames.map((cacheName) => {
          if (cacheName !== CACHE_VERSION && cacheName !== PHOTO_CACHE) {
            return caches.delete(cacheName);
          }
        })
      );
    })
  );
  self.clients.claim();
});

// Fetch: network-first for photos, cache-first for static
self.addEventListener('fetch', (event) => {
  const url = new URL(event.request.url);

  // Network-first for photo API (prioritize fresh photos)
  if (url.pathname.startsWith('/api/photo')) {
    event.respondWith(
      fetch(event.request)
        .then((response) => {
          const responseClone = response.clone();
          caches.open(PHOTO_CACHE).then((cache) => {
            cache.put(event.request, responseClone);
          });
          return response;
        })
        .catch(() => {
          // Network failed - return cached version if available
          return caches.match(event.request);
        })
    );
  }
  // Cache-first for static assets
  else {
    event.respondWith(
      caches.match(event.request).then((response) => {
        return response || fetch(event.request);
      })
    );
  }
});
```

Register Service Worker in `public/slideshow.html` by adding this script before closing `</body>` tag (after existing scripts):

```javascript
// Register Service Worker
if ('serviceWorker' in navigator) {
  navigator.serviceWorker.register('/sw.js')
    .then((registration) => {
      console.log('Service Worker registered:', registration.scope);
    })
    .catch((error) => {
      console.error('Service Worker registration failed:', error);
    });
}
```

IMPORTANT: Network-first strategy means photos are always fresh when online, but slideshow continues working if network drops. Cache-first for static assets means faster loads on repeat visits.
